def baca_csv_bytes(data_bytes):
    for enc in ("utf-8-sig", "utf-8", "cp1252", "latin-1"):
        try:
            return pd.read_csv(io.BytesIO(data_bytes), encoding=enc, low_memory=False)
        except UnicodeDecodeError:
            continue
    raise ValueError("Encoding CSV tidak dikenali.")


def baca_dataset_upload(file_upload):
    """Membaca CSV/XLSX langsung atau dalam ZIP/TAR/RAR/7Z."""
    nama = file_upload.name.lower()
    isi = file_upload.getvalue()
    hasil = {}

    def tambah(nama_internal, data_bytes):
        n = nama_internal.lower()
        if n.endswith(".csv"):
            hasil[nama_internal] = baca_csv_bytes(data_bytes)
        elif n.endswith(".xlsx"):
            hasil[nama_internal] = pd.read_excel(io.BytesIO(data_bytes))

    if nama.endswith(".csv"):
        hasil[Path(nama).name] = baca_csv_bytes(isi)
        return hasil
    if nama.endswith(".xlsx"):
        hasil[Path(nama).name] = pd.read_excel(io.BytesIO(isi))
        return hasil

    if zipfile.is_zipfile(io.BytesIO(isi)):
        with zipfile.ZipFile(io.BytesIO(isi)) as arsip:
            for item in arsip.infolist():
                if item.is_dir() or "__MACOSX/" in item.filename:
                    continue
                if Path(item.filename).name.startswith("._"):
                    continue
                if item.filename.lower().endswith((".csv", ".xlsx")):
                    tambah(item.filename, arsip.read(item))
        if hasil:
            return hasil

    if tarfile.is_tarfile(io.BytesIO(isi)):
        with tarfile.open(fileobj=io.BytesIO(isi), mode="r:*") as arsip:
            for item in arsip.getmembers():
                if not item.isfile():
                    continue
                if item.name.lower().endswith((".csv", ".xlsx")):
                    berkas = arsip.extractfile(item)
                    if berkas:
                        tambah(item.name, berkas.read())
        if hasil:
            return hasil

    if nama.endswith(".rar"):
        try:
            import rarfile
            with rarfile.RarFile(io.BytesIO(isi)) as arsip:
                for item in arsip.infolist():
                    if not item.is_dir() and item.filename.lower().endswith((".csv", ".xlsx")):
                        with arsip.open(item) as berkas:
                            tambah(item.filename, berkas.read())
            if hasil:
                return hasil
        except Exception as exc:
            raise ValueError(f"Gagal membaca file RAR: {exc}")

    if nama.endswith(".7z"):
        try:
            import py7zr
            with py7zr.SevenZipFile(io.BytesIO(isi), mode="r") as arsip:
                nama_file = arsip.getnames()
                data_file = arsip.read(nama_file)
                for n, obj in data_file.items():
                    if n.lower().endswith((".csv", ".xlsx")):
                        tambah(n, obj.read())
            if hasil:
                return hasil
        except Exception as exc:
            raise ValueError(f"Gagal membaca file 7Z: {exc}")

    raise ValueError("Tidak ditemukan berkas CSV atau format tidak didukung.")


def nama_file_sederhana(nama):
    return Path(nama.replace("\\", "/")).name.lower()


def cari_tabel(kumpulan_data, nama_target):
    target = nama_target.lower()
    cocok = [(n, d) for n, d in kumpulan_data.items() if nama_file_sederhana(n) == target]
    if not cocok:
        return None
    return cocok[0][1].copy()


def integrasikan_opentender(kumpulan_data):
    """Integrasi relasional main.csv + awards.csv atau format file tunggal."""
    main = cari_tabel(kumpulan_data, "main.csv")
    awards = cari_tabel(kumpulan_data, "awards.csv")

    if main is None and awards is None:
        kandidat = next(iter(kumpulan_data.values())).copy()
        wajib = {"tender_minValue_amount", "value_amount", "tender_mainProcurementCategory"}
        if wajib.issubset(kandidat.columns):
            return kandidat
        raise ValueError(
            "File tunggal harus memiliki: tender_minValue_amount, value_amount, tender_mainProcurementCategory."
        )

    if main is None or awards is None:
        raise ValueError("Data relasional OpenTender membutuhkan main.csv dan awards.csv.")

    wajib_main = ["id", "ocid", "tender_minValue_amount", "tender_mainProcurementCategory"]
    wajib_awards = ["value_amount", "main_id", "main_ocid"]
    kurang_m = [c for c in wajib_main if c not in main.columns]
    kurang_a = [c for c in wajib_awards if c not in awards.columns]
    if kurang_m or kurang_a:
        raise ValueError(f"Kolom hilang: main={kurang_m}, awards={kurang_a}")

    for col in ("id", "ocid"):
        main[col] = main[col].astype("string").str.strip()
    for col in ("main_id", "main_ocid"):
        awards[col] = awards[col].astype("string").str.strip()

    id_m = main.loc[main["id"].notna() & main["id"].ne(""), "id"]
    id_a = awards.loc[awards["main_id"].notna() & awards["main_id"].ne(""), "main_id"]
    oc_m = main.loc[main["ocid"].notna() & main["ocid"].ne(""), "ocid"]
    oc_a = awards.loc[awards["main_ocid"].notna() & awards["main_ocid"].ne(""), "main_ocid"]

    cocok_id = len(set(id_m.unique()) & set(id_a.unique()))
    cocok_ocid = len(set(oc_m.unique()) & set(oc_a.unique()))
    if cocok_id >= cocok_ocid and cocok_id > 0:
        k_main, k_awards = "id", "main_id"
    elif cocok_ocid > 0:
        k_main, k_awards = "ocid", "main_ocid"
    else:
        raise ValueError("Tidak ditemukan ID/OCID yang berelasi antara main.csv dan awards.csv.")

    main["tender_minValue_amount"] = pd.to_numeric(main["tender_minValue_amount"], errors="coerce")
    awards["value_amount"] = pd.to_numeric(awards["value_amount"], errors="coerce")

    hasil = awards.merge(
        main, left_on=k_awards, right_on=k_main, how="left",
        suffixes=("_award", "_main")
    )
    if hasil.empty:
        raise ValueError("Hasil penggabungan dataset kosong.")
    return hasil


def deteksi_anomali(data_baru, paket):
    """Inference Isolation Forest dengan preprocessing 7-fitur identik saat training."""
    wajib = ["tender_minValue_amount", "value_amount", "tender_mainProcurementCategory"]
    kurang = [c for c in wajib if c not in data_baru.columns]
    if kurang:
        raise ValueError("Kolom model belum tersedia: " + ", ".join(kurang))

    data_baru = data_baru.copy()
    nilai_min = pd.to_numeric(data_baru["tender_minValue_amount"], errors="coerce")
    nilai_kontrak = pd.to_numeric(data_baru["value_amount"], errors="coerce")

    fitur = pd.DataFrame(index=data_baru.index)
    fitur["log_min_value"] = np.log1p(nilai_min.clip(lower=0))
    fitur["log_award_value"] = np.log1p(nilai_kontrak.clip(lower=0))
    fitur["selisih_award_rel"] = ((nilai_min - nilai_kontrak) / nilai_min).where(nilai_min.ne(0))
    fitur = fitur.replace([np.inf, -np.inf], np.nan)

    fitur_imp = pd.DataFrame(
        paket["imputer"].transform(fitur),
        columns=fitur.columns,
        index=data_baru.index,
    )
    fitur_imp["min_value_missing"] = nilai_min.isna().astype(int)

    kategori = data_baru["tender_mainProcurementCategory"].astype("string").str.strip().str.lower()
    for item in ("goods", "services", "works"):
        fitur_imp[f"tender_mainProcurementCategory_{item}"] = (
            kategori == item
        ).fillna(False).astype(int)

    fitur_model = fitur_imp.reindex(columns=paket["fitur_model"])
    prediksi = paket["model"].predict(fitur_model)
    skor = paket["model"].score_samples(fitur_model)

    hasil = data_baru.copy()
    hasil["log_min_value"] = fitur["log_min_value"].values
    hasil["log_award_value"] = fitur["log_award_value"].values
    hasil["selisih_award_rel"] = fitur["selisih_award_rel"].values
    hasil["prediksi_anomali"] = prediksi
    hasil["skor_anomali"] = -skor
    hasil["status_deteksi"] = np.where(prediksi == -1, "Anomali", "Normal")
    return hasil

"""
Modul Firebase Realtime Database untuk proyek OpenTender Anomaly Detection.

Fungsi utama:
    - init_firebase()            : inisialisasi koneksi Firebase (sekali saja)
    - test_connection()          : uji koneksi Firebase

    - ambil_riwayat_firebase()   : baca riwayat analysis_history untuk dashboard
    - ambil_dashboard_firebase() : baca node dashboard

Prinsip keamanan:
    - Tidak menghapus node historis yang sudah ada.
    - Hanya MENAMBAH atau MEMPERBARUI node.
    - Jika Firebase gagal, MySQL tidak akan dipengaruhi (dipanggil setelah MySQL sukses).

import threading
from datetime import datetime, timezone
from typing import Optional

import firebase_admin
from firebase_admin import credentials, db

from config import FIREBASE_CREDENTIALS_PATH, FIREBASE_DATABASE_URL

# Lock agar inisialisasi aman di environment multi-thread (Streamlit)
_init_lock = threading.Lock()
_initialized = False


# ============================================================
# INISIALISASI
# ============================================================

def init_firebase() -> bool:
    """
    Menginisialisasi koneksi Firebase Admin SDK.
    Aman dipanggil berkali-kali — hanya dijalankan sekali.
    Mengembalikan True jika berhasil, False jika gagal.
    """
    global _initialized
    with _init_lock:
        if _initialized:
            return True
        try:
            # Jika sudah ada app yang diinisialisasi sebelumnya
            if firebase_admin._apps:
                _initialized = True
                return True

            cred = credentials.Certificate(FIREBASE_CREDENTIALS_PATH)
            firebase_admin.initialize_app(cred, {"databaseURL": FIREBASE_DATABASE_URL})
            _initialized = True
            return True
        except Exception as e:
            print(f"[Firebase] Inisialisasi gagal: {e}")
            return False


def test_connection() -> tuple[bool, str]:
    """
    Menguji kesiapan koneksi Firebase Admin SDK.
    Mengembalikan (True, pesan) jika siap, (False, pesan error) jika gagal.
    """
    try:
        if not init_firebase():
            return False, "Inisialisasi Firebase gagal."
        if firebase_admin._apps:
            return True, "Koneksi Firebase terhubung."
        return False, "Firebase belum terinisialisasi."
    except Exception as e:
        return False, f"Koneksi Firebase gagal: {e}"


# ============================================================
# HELPER
# ============================================================

def _now_iso() -> str:
    return datetime.now(tz=timezone.utc).isoformat()


def _safe_dict(obj) -> dict:
    """Pastikan semua nilai dalam dict bisa diserialisasi ke JSON Firebase."""
    if not isinstance(obj, dict):
        return {}
    result = {}
    for k, v in obj.items():
        if isinstance(v, dict):
            result[str(k)] = _safe_dict(v)
        elif isinstance(v, list):
            result[str(k)] = [str(x) for x in v]
        elif v is None or v != v:  # None atau NaN
            result[str(k)] = None
        else:
            result[str(k)] = v
    return result


# ============================================================
# KIRIM RINGKASAN BATCH
# ============================================================

def kirim_ringkasan_batch(
    batch_id: str,
    filename: str,
    total_rows: int,
    total_normal: int,
    total_anomali: int,
    catatan: Optional[str] = None,
) -> tuple[bool, str]:
    """
    Menyimpan ringkasan hasil satu batch upload ke:
        opentender/analysis_history/{batch_id}

    Tidak menimpa atau menghapus batch lain.
    """
    try:
        if not init_firebase():
            return False, "Firebase tidak terinisialisasi."

        total = total_normal + total_anomali
        pct   = round(total_anomali / total * 100, 2) if total > 0 else 0.0

        payload = {
            "batch_id":       batch_id,
            "filename":       str(filename),
            "total_rows":     int(total_rows),
            "total_normal":   int(total_normal),
            "total_anomali":  int(total_anomali),
            "pct_anomali":    float(pct),
            "status":         "done",
            "timestamp":      _now_iso(),
        }
        if catatan:
            payload["catatan"] = str(catatan)

        ref = db.reference(f"opentender/analysis_history/{batch_id}")
        ref.set(payload)
        return True, f"Ringkasan batch '{batch_id}' berhasil dikirim."
    except Exception as e:
        return False, f"Gagal kirim ringkasan batch: {e}"

def perbarui_dashboard(
    total_normal_baru: int,
    total_anomali_baru: int,
) -> tuple[bool, str]:
    """
    Memperbarui node opentender/dashboard dengan menambahkan
    hasil batch terbaru ke akumulasi yang sudah ada.

    Node yang diperbarui:
        opentender/dashboard/total_upload_tenders
        opentender/dashboard/total_upload_normal
        opentender/dashboard/total_upload_anomali
        opentender/dashboard/pct_upload_anomali
        opentender/dashboard/last_updated

    Node historis (total_tenders, total_awards, dll.) TIDAK disentuh.
    """
    try:
        if not init_firebase():
            return False, "Firebase tidak terinisialisasi."

        ref = db.reference("opentender/dashboard")
        snapshot = ref.get() or {}

        # Akumulasi khusus untuk hasil upload (terpisah dari data historis)
        prev_normal  = int(snapshot.get("total_upload_normal",  0) or 0)
        prev_anomali = int(snapshot.get("total_upload_anomali", 0) or 0)

        new_normal  = prev_normal  + total_normal_baru
        new_anomali = prev_anomali + total_anomali_baru
        new_total   = new_normal + new_anomali
        new_pct     = round(new_anomali / new_total * 100, 2) if new_total > 0 else 0.0

        ref.update({
            "total_upload_tenders": int(new_total),
            "total_upload_normal":  int(new_normal),
            "total_upload_anomali": int(new_anomali),
            "pct_upload_anomali":   float(new_pct),
            "last_updated":         _now_iso(),
        })
        return True, "Node dashboard berhasil diperbarui."
    except Exception as e:
        return False, f"Gagal memperbarui dashboard Firebase: {e}"

def _safe(val):
    """Konversi nilai numpy/pandas ke tipe Python biasa untuk MySQL."""
    if val is None:
        return None
    if isinstance(val, (float, np.floating)):
        return None if (np.isnan(val) or np.isinf(val)) else float(val)
    if isinstance(val, (int, np.integer)):
        return int(val)
    s = str(val).strip()
    return None if s == "" or s.lower() == "nan" else s


def simpan_hasil_prediksi(
    conn,
    batch_id: str,
    df_hasil: pd.DataFrame,
    batch_size: int = MYSQL_BATCH_SIZE,
) -> tuple[int, int]:
    """
    Menyimpan DataFrame hasil prediksi ke tabel analysis_results.
    Mengembalikan (total_normal, total_anomali).
    """
    sql = """
        INSERT INTO analysis_results (
            batch_id,
            tender_id, tender_ocid,
            tender_datePublished,
            tender_mainProcurementCategory,
            tender_minValue_amount,
            value_amount,
            log_min_value, log_award_value, selisih_award_rel,
            skor_anomali,
            prediksi_anomali,
            status_deteksi
        ) VALUES (
            %s,
            %s, %s,
            %s,
            %s,
            %s,
            %s,
            %s, %s, %s,
            %s,
            %s,
            %s
        )
    """

    def _get(row, *keys, default=None):
        for k in keys:
            if k in row.index and pd.notna(row[k]):
                return row[k]
        return default

    total_normal  = 0
    total_anomali = 0
    total_saved   = 0

    cursor = conn.cursor()

    for start in range(0, len(df_hasil), batch_size):
        chunk = df_hasil.iloc[start : start + batch_size]
        records = []

        for _, row in chunk.iterrows():
            status = str(row.get("status_deteksi", "")) if "status_deteksi" in row.index else ""
            if status == "Normal":
                total_normal += 1
            elif status == "Anomali":
                total_anomali += 1

            # Parse tanggal
            tgl = _get(row, "tender_datePublished", "date")
            if tgl is not None:
                try:
                    tgl = pd.to_datetime(tgl, utc=True).tz_localize(None).to_pydatetime()
                except Exception:
                    tgl = None

            records.append((
                batch_id,
                _safe(_get(row, "id", "main_id", "id_main")),
                _safe(_get(row, "ocid", "main_ocid", "ocid_main")),
                tgl,
                _safe(_get(row, "tender_mainProcurementCategory")),
                _safe(_get(row, "tender_minValue_amount")),
                _safe(_get(row, "value_amount")),
                _safe(_get(row, "log_min_value")),
                _safe(_get(row, "log_award_value")),
                _safe(_get(row, "selisih_award_rel")),
                _safe(_get(row, "skor_anomali")),
                _safe(_get(row, "prediksi_anomali")),
                status if status else None,
            ))

        try:
            cursor.executemany(sql, records)
            conn.commit()
            total_saved += len(records)
        except Exception:
            conn.rollback()
            cursor.close()
            raise

    cursor.close()
    return total_normal, total_anomali

import sys

import joblib

def test_pipeline():
    print("=" * 60)
    print("PENGUJIAN INTEGRASI END-TO-END (CSV KECIL)")
    print("=" * 60)

    # 1. Pemuatan Model
    print("\n[1] Memuat model...")
    paket = joblib.load(str(FILE_MODEL))
    print("    Model berhasil dimuat:", type(paket["model"]).__name__)

    # 2. Baca CSV Uji
    print("\n[2] Membaca dataset uji kecil...")
    main_df = pd.read_csv("dataset_test/main.csv")
    awards_df = pd.read_csv("dataset_test/awards.csv")
    print(f"    main.csv   : {len(main_df)} baris")
    print(f"    awards.csv : {len(awards_df)} baris")

    # 3. Integrasi
    print("\n[3] Mengintegrasikan dataset (merge)...")
    for col in ("id", "ocid"):
        main_df[col] = main_df[col].astype("string").str.strip()
    for col in ("main_id", "main_ocid"):
        awards_df[col] = awards_df[col].astype("string").str.strip()

    main_df["tender_minValue_amount"] = pd.to_numeric(main_df["tender_minValue_amount"], errors="coerce")
    awards_df["value_amount"] = pd.to_numeric(awards_df["value_amount"], errors="coerce")

    merged = awards_df.merge(
        main_df, left_on="main_id", right_on="id", how="left",
        suffixes=("_award", "_main")
    )
    print(f"    Hasil merge: {len(merged)} baris")

    # 4. Deteksi Anomali
    print("\n[4] Menjalankan deteksi anomali...")
    nilai_min = pd.to_numeric(merged["tender_minValue_amount"], errors="coerce")
    nilai_kontrak = pd.to_numeric(merged["value_amount"], errors="coerce")

    fitur = pd.DataFrame(index=merged.index)
    fitur["log_min_value"] = np.log1p(nilai_min.clip(lower=0))
    fitur["log_award_value"] = np.log1p(nilai_kontrak.clip(lower=0))
    fitur["selisih_award_rel"] = ((nilai_min - nilai_kontrak) / nilai_min).where(nilai_min.ne(0))
    fitur = fitur.replace([np.inf, -np.inf], np.nan)

    fitur_imp = pd.DataFrame(
        paket["imputer"].transform(fitur),
        columns=fitur.columns,
        index=merged.index,
    )
    fitur_imp["min_value_missing"] = nilai_min.isna().astype(int)

    kategori = merged["tender_mainProcurementCategory"].astype("string").str.strip().str.lower()
    for item in ("goods", "services", "works"):
        fitur_imp[f"tender_mainProcurementCategory_{item}"] = (
            kategori == item
        ).fillna(False).astype(int)

    fitur_model = fitur_imp.reindex(columns=paket["fitur_model"])
    pred = paket["model"].predict(fitur_model)
    skor = paket["model"].score_samples(fitur_model)

    hasil = merged.copy()
    hasil["log_min_value"] = fitur["log_min_value"].values
    hasil["log_award_value"] = fitur["log_award_value"].values
    hasil["selisih_award_rel"] = fitur["selisih_award_rel"].values
    hasil["prediksi_anomali"] = pred
    hasil["skor_anomali"] = -skor
    hasil["status_deteksi"] = np.where(pred == -1, "Anomali", "Normal")

    total_rows = len(hasil)
    total_anomali = int((hasil["status_deteksi"] == "Anomali").sum())
    total_normal = total_rows - total_anomali
    pct_anomali = round((total_anomali / total_rows) * 100, 2)

    print(f"    Total dianalisis : {total_rows}")
    print(f"    Normal           : {total_normal}")
    print(f"    Anomali          : {total_anomali} ({pct_anomali}%)")

    # 5. Simpan ke MySQL
    batch_id = f"test-{uuid.uuid4().hex[:8]}"
    print(f"\n[5] Menyimpan ke MySQL (Batch ID: {batch_id})...")
    conn = get_connection()
    simpan_batch(conn, batch_id, "test_dataset_small.csv", total_rows=total_rows, status="processing")
    n_norm, n_anom = simpan_hasil_prediksi(conn, batch_id, hasil)
    perbarui_status_batch(conn, batch_id, n_norm, n_anom, status="done")

    # Verifikasi langsung dari MySQL

    cur.execute("SELECT * FROM upload_batches WHERE batch_id = %s", (batch_id,))
    batch_row = cur.fetchone()
    print("    Verifikasi batch MySQL:", batch_row)

    cur.execute("SELECT COUNT(*) AS total FROM analysis_results WHERE batch_id = %s", (batch_id,))
    res_count = cur.fetchone()["total"]
    print(f"    Verifikasi baris analysis_results: {res_count} baris tersimpan.")
    cur.close()
    conn.close()

    # 6. Kirim ke Firebase RTDB
    print(f"\n[6] Mengirim ke Firebase Realtime Database...")
    ok_fb, msg_fb = db_firebase.kirim_ringkasan_batch(
        batch_id=batch_id,
        filename="test_dataset_small.csv",
        total_rows=total_rows,
        total_normal=total_normal,
        total_anomali=total_anomali,
        catatan="pengujian_otomatis",
    )
    print("    Kirim history:", ok_fb, msg_fb)

    ok_dash, msg_dash = db_firebase.perbarui_dashboard(total_normal, total_anomali)
    print("    Update dashboard:", ok_dash, msg_dash)

    riwayat = db_firebase.ambil_riwayat_firebase(limit=5)
    print("    Riwayat terbaru di Firebase:")
    for r in riwayat:

            print(f"    -> DITEMUKAN BATCH {batch_id}:", r)

def convert_value(value, data_type):

    if pd.isna(value) or str(value).strip() == "":
        return None

    value = str(value).strip()

    if data_type in (
        "tinyint", "smallint", "mediumint",
        "int", "integer", "bigint"
    ):
        return int(float(value))

    if data_type in ("float", "double", "real"):
        return float(value)

    if data_type in ("decimal", "numeric"):
        return Decimal(value)

    if data_type == "date":
        return pd.to_datetime(value, utc=True).date()

    if data_type in ("datetime", "timestamp"):
        dt = pd.to_datetime(value, utc=True)
        return dt.tz_localize(None).to_pydatetime()

    return value

def get_table_columns(cursor, table):

    cursor.execute("""
        SELECT COLUMN_NAME, DATA_TYPE, EXTRA
        FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_SCHEMA = %s
        AND TABLE_NAME = %s
        ORDER BY ORDINAL_POSITION
    """, (DB_CONFIG["database"], table))

    columns = cursor.fetchall()

    if not columns:
        raise Exception(f"Tabel {table} tidak ditemukan.")

    # Abaikan kolom AUTO_INCREMENT dan generated
    result = []

    for column, data_type, extra in columns:

        if "auto_increment" in extra.lower():
            continue

        if "generated" in extra.lower():
            continue

        result.append((column, data_type))

    return result

def check_empty_database(cursor):

def import_csv(connection, filename, table):

    path = os.path.join(DATASET_DIR, filename)

    if not os.path.exists(path):
        raise FileNotFoundError(f"File tidak ditemukan: {path}")

    cursor = connection.cursor()

    print("\n" + "=" * 55)
    print(f"MIGRASI: {filename} -> {table}")
    print("=" * 55)

    columns = get_table_columns(cursor, table)

    db_column_names = [col[0] for col in columns]

    # Periksa kesesuaian kolom CSV dan MySQL
    csv_columns = list(pd.read_csv(path, nrows=0).columns)

    missing = set(db_column_names) - set(csv_columns)
    extra = set(csv_columns) - set(db_column_names)

    if missing or extra:
        raise Exception(
            f"Kolom tidak sesuai pada {table}.\n"
            f"Kolom yang hilang: {missing}\n"
            f"Kolom tambahan: {extra}"
        )

    column_names = ", ".join(
        f"`{name}`" for name in db_column_names
    )

    placeholders = ", ".join(
        ["%s"] * len(db_column_names)
    )

    sql = (
        f"INSERT INTO `{table}` "
        f"({column_names}) VALUES ({placeholders})"
    )

    total_inserted = 0

    for chunk in pd.read_csv(
        path,
        dtype=str,
        chunksize=CHUNK_SIZE,
        keep_default_na=True
    ):

        records = []

        for row in chunk[db_column_names].itertuples(
            index=False,
            name=None
        ):

            converted = []

            for value, (_, data_type) in zip(row, columns):
                converted.append(
                    convert_value(value, data_type)
                )

            records.append(tuple(converted))

        try:
            cursor.executemany(sql, records)
            connection.commit()

        except Exception:
            connection.rollback()
            raise

        total_inserted += len(records)

        print(
            f"Berhasil memasukkan: "
            f"{total_inserted:,} baris",
            end="\r"
        )

    cursor.close()

    print(f"\nSELESAI: {total_inserted:,} baris")

    return total_inserted

BASE_DIR = Path(__file__).resolve().parent.parent

KEY_PATH = BASE_DIR / "credentials" / "firebase-key.json"

ANOMALY_FILE = BASE_DIR / "hasil_deteksi_anomali_opentender.csv"

DATABASE_URL = (
    "https://opentender-bigdata-default-rtdb.firebaseio.com/"
)

MYSQL_CONFIG = {
    "host": "localhost",
    "port": 3306,
    "user": "root",
    "password": "",
    "database": "opentender_bigdata"
}


# ==========================================
# 2. KONEKSI MYSQL
# ==========================================

def koneksi_mysql():
    return mysql.connector.connect(**MYSQL_CONFIG)


# ==========================================
# 3. FUNGSI MENJALANKAN QUERY
# ==========================================

def ambil_data(cursor, query):

    cursor.execute(query)

    kolom = [item[0] for item in cursor.description]

    hasil = []

    for row in cursor.fetchall():
        hasil.append(dict(zip(kolom, row)))

    return hasil


# ==========================================
# 4. KONEKSI FIREBASE RTDB
# ==========================================

def koneksi_firebase():

    if not KEY_PATH.exists():
        raise FileNotFoundError(
            f"File Firebase tidak ditemukan: {KEY_PATH}"
        )

    cred = credentials.Certificate(str(KEY_PATH))

    firebase_admin.initialize_app(
        cred,
        {
            "databaseURL": DATABASE_URL
        }
    )

    print("Koneksi Firebase Realtime Database berhasil!")


# ==========================================
# 5. MENGHITUNG STATISTIK ANOMALI
# ==========================================

def hitung_statistik_anomali():

    if not ANOMALY_FILE.exists():
        raise FileNotFoundError(
            f"File hasil deteksi tidak ditemukan: {ANOMALY_FILE}"
        )

    total_normal = 0
    total_anomali = 0

    print("Membaca hasil deteksi Isolation Forest...")

    # Membaca CSV secara bertahap agar hemat RAM
    for chunk in pd.read_csv(
        ANOMALY_FILE,
        usecols=["status_deteksi"],
        chunksize=100000
    ):

        jumlah = chunk["status_deteksi"].value_counts()

        total_normal += int(jumlah.get("Normal", 0))
        total_anomali += int(jumlah.get("Anomali", 0))

    total_data = total_normal + total_anomali

    persentase_anomali = (
        round((total_anomali / total_data) * 100, 2)
        if total_data > 0 else 0
    )

    return {
        "total_data_dianalisis": total_data,
        "total_anomalies": total_anomali,
        "normal_data": total_normal,
        "anomaly_percentage": persentase_anomali
    }


# ==========================================
# 6. PROSES PENGOLAHAN DATA MYSQL
# ==========================================

def proses_statistik(cursor):

    print("\nMenghitung total data...")

    total_tenders = ambil_data(
        cursor,
        "SELECT COUNT(*) AS total FROM tenders"
    )[0]["total"]

    total_awards = ambil_data(
        cursor,
        "SELECT COUNT(*) AS total FROM awards"
    )[0]["total"]

    total_suppliers = ambil_data(
        cursor,
        "SELECT COUNT(*) AS total FROM award_suppliers"
    )[0]["total"]


    print("Menghitung statistik tender per tahun...")

    tenders_by_year = ambil_data(
        cursor,
        """
        SELECT
            LEFT(tender_datePublished, 4) AS tahun,
            COUNT(*) AS total
        FROM tenders
        WHERE tender_datePublished IS NOT NULL
          AND LENGTH(tender_datePublished) >= 4
        GROUP BY LEFT(tender_datePublished, 4)
        ORDER BY tahun
        """
    )


    print("Menghitung statistik berdasarkan status...")

    tenders_by_status = ambil_data(
        cursor,
        """
        SELECT
            COALESCE(
                tender_status,
                'Tidak diketahui'
            ) AS status,
            COUNT(*) AS total
        FROM tenders
        GROUP BY tender_status
        ORDER BY total DESC
        """
    )


    print("Menghitung 10 buyer teratas...")

    top_buyers = ambil_data(
        cursor,
        """
        SELECT
            COALESCE(
                buyer_name,
                'Tidak diketahui'
            ) AS nama,
            COUNT(*) AS total_tender
        FROM tenders
        GROUP BY buyer_name
        ORDER BY total_tender DESC
        LIMIT 10
        """
    )


    print("Menghitung 10 supplier teratas...")

    top_suppliers = ambil_data(
        cursor,
        """
        SELECT
            COALESCE(
                name,
                'Tidak diketahui'
            ) AS nama,
            COUNT(*) AS total_keterlibatan
        FROM award_suppliers
        GROUP BY name
        ORDER BY total_keterlibatan DESC
        LIMIT 10
        """
    )


    print("Menghitung tender berdasarkan kategori...")

    tenders_by_category = ambil_data(
        cursor,
        """
        SELECT
            COALESCE(
                tender_mainProcurementCategory,
                'Tidak diketahui'
            ) AS kategori,
            COUNT(*) AS total
        FROM tenders
        GROUP BY tender_mainProcurementCategory
        ORDER BY total DESC
        """
    )


    print("Menghitung instansi pengadaan teratas...")

    top_procuring_entities = ambil_data(
        cursor,
        """
        SELECT
            COALESCE(
                tender_procuringEntity_name,
                'Tidak diketahui'
            ) AS nama,
            COUNT(*) AS total_tender
        FROM tenders
        GROUP BY tender_procuringEntity_name
        ORDER BY total_tender DESC
        LIMIT 10
        """
    )


    print("Menghitung awards berdasarkan tahun...")

    awards_by_year = ambil_data(
        cursor,
        """
        SELECT
            LEFT(date, 4) AS tahun,
            COUNT(*) AS total
        FROM awards
        WHERE date IS NOT NULL
          AND LENGTH(date) >= 4
        GROUP BY LEFT(date, 4)
        ORDER BY tahun
        """
    )


    print("Menghitung awards berdasarkan mata uang...")

    awards_by_currency = ambil_data(
        cursor,
        """
        SELECT
            COALESCE(
                value_currency,
                'Tidak diketahui'
            ) AS mata_uang,
            COUNT(*) AS total
        FROM awards
        GROUP BY value_currency
        ORDER BY total DESC
        """
    )


    print("Menghitung keterlibatan supplier per tahun...")

    suppliers_by_year = ambil_data(
        cursor,
        """
        SELECT
            LEFT(t.tender_datePublished, 4) AS tahun,
            COUNT(*) AS total
        FROM award_suppliers s
        JOIN tenders t
            ON s.main_ocid = t.ocid
        WHERE t.tender_datePublished IS NOT NULL
          AND LENGTH(t.tender_datePublished) >= 4
        GROUP BY LEFT(t.tender_datePublished, 4)
        ORDER BY tahun
        """
    )


    # ======================================
    # STATISTIK ISOLATION FOREST
    # ======================================

    print("Menghitung hasil deteksi anomali...")

    anomaly_stats = hitung_statistik_anomali()


    # ======================================
    # MENYUSUN HASIL STATISTIK
    # ======================================

    return {

        "dashboard": {
            "total_tenders": int(total_tenders),
            "total_awards": int(total_awards),
            "total_suppliers": int(total_suppliers),
            "total_anomalies": anomaly_stats["total_anomalies"]
        },

        "statistics": {

            "tenders_by_year": tenders_by_year,

            "tenders_by_status": tenders_by_status,

            "top_buyers": top_buyers,

            "top_suppliers": top_suppliers,

            "tenders_by_category": tenders_by_category,

            "top_procuring_entities": top_procuring_entities,

            "awards_by_year": awards_by_year,

            "awards_by_currency": awards_by_currency,

            "suppliers_by_year": suppliers_by_year,

            "anomaly_statistics": anomaly_stats
        },

        "metadata": {
            "source": "OpenTender Indonesia - MySQL + Isolation Forest",
            "generated_at": datetime.now().astimezone().isoformat()
        }
    }

def main():

    koneksi = None
    cursor = None

    try:

        print("====================================")
        print(" OPEN TENDER - MYSQL TO FIREBASE")
        print("====================================")

        koneksi = koneksi_mysql()

        cursor = koneksi.cursor()

        print("Koneksi MySQL berhasil!")

        koneksi_firebase()

        data = proses_statistik(cursor)

        print("\nMengirim data ke Firebase...")

        kirim_ke_firebase(data)

        print("\n====================================")
        print("MIGRASI STATISTIK BERHASIL!")
        print("====================================")

        print(
            "Total Tender:",
            data["dashboard"]["total_tenders"]
        )

        print(
            "Total Awards:",
            data["dashboard"]["total_awards"]
        )

        print(
            "Total Suppliers:",
            data["dashboard"]["total_suppliers"]
        )

        print(
            "Total Anomali:",
            data["dashboard"]["total_anomalies"]
        )

        print(
            "Data Normal:",
            data["statistics"]["anomaly_statistics"]["normal_data"]
        )

        print(
            "Persentase Anomali:",
            str(
                data["statistics"]["anomaly_statistics"][
                    "anomaly_percentage"
                ]
            ) + "%"
        )

        print("\nNode yang diperbarui:")
        print("- opentender/dashboard")
        print("- opentender/statistics")
        print("- opentender/metadata")

        print("\n====================================")

    except Exception as error:

        print("\nPROSES GAGAL!")
        print("Error:", error)

    finally:

        if cursor is not None:
            cursor.close()

        if koneksi is not None and koneksi.is_connected():
            koneksi.close()


if __name__ == "__main__":
    main()

import pandas as pd
import sqlite3
import os

DATASET = "dataset"
DB_TEMP = "validasi_sementara.sqlite"
CHUNK_SIZE = 100000

if os.path.exists(DB_TEMP):
    os.remove(DB_TEMP)

db = sqlite3.connect(DB_TEMP)
cursor = db.cursor()

# Tabel sementara untuk pemeriksaan ID
cursor.executescript("""
CREATE TABLE main_ids (
    ocid TEXT,
    main_id TEXT
);

CREATE TABLE award_ids (
    id TEXT,
    main_ocid TEXT,
    main_id TEXT
);

CREATE TABLE supplier_ids (
    id TEXT,
    main_ocid TEXT,
    main_id TEXT,
    awards_id TEXT
);

CREATE INDEX idx_main_ocid ON main_ids(ocid);
CREATE INDEX idx_main_id ON main_ids(main_id);
CREATE INDEX idx_award_id ON award_ids(id);
""")


def load_csv(filename, columns, table):
    path = os.path.join(DATASET, filename)
    total = 0

    for chunk in pd.read_csv(
        path,
        usecols=columns,
        dtype=str,
        chunksize=CHUNK_SIZE
    ):
        # PENTING: Pastikan urutan kolom sesuai tabel SQLite
        chunk = chunk[columns]

        records = [
            tuple(None if pd.isna(v) else v for v in row)
            for row in chunk.itertuples(index=False, name=None)
        ]

        placeholders = ",".join(["?"] * len(columns))

        cursor.executemany(
            f"INSERT INTO {table} VALUES ({placeholders})",
            records
        )

        total += len(records)
        db.commit()

    print(f"{filename}: {total:,} baris")


print("MEMBACA DATASET...")
load_csv("main.csv", ["ocid", "id"], "main_ids")
load_csv("awards.csv", ["id", "main_ocid", "main_id"], "award_ids")
load_csv(
    "awards_suppliers.csv",
    ["id", "main_ocid", "main_id", "awards_id"],
    "supplier_ids"
)
